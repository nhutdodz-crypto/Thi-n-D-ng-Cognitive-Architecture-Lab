#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ThoughtVector {
    pub id:               Uuid,        // UUID v7 (timestamp-ordered)
    pub session_id:       Uuid,
    pub platform:         Platform,
    pub raw_thought:      String,      // Immutable after creation
    pub complexity_score: f32,         // [0.0, 1.0] computed by scorer
    pub intent_embedding: Vec<f16>,     // 1536-dim, 3KB, pgvector-compatible
    pub fidelity_target:  u8,          // Minimum acceptable score [0,100]
    pub timestamp_ns:     u64,         // Nanosecond precision
    pub language:         String,      // "vi" | "en" | "bilingual"
}

impl ThoughtVector {
    pub fn new(session: Uuid, platform: Platform, thought: impl Into<String>) -> Self {
        Self {
            id:               Uuid::now_v7(),
            session_id:       session,
            platform,
            raw_thought:      thought.into(),
            complexity_score: 0.0,
            intent_embedding: Vec::with_capacity(EMBEDDING_DIM),
            fidelity_target:  FIDELITY_THRESHOLD,
            timestamp_ns:     now_ns(),
            language:         "bilingual".into(),
        }
    }

    #[inline]
    pub fn on_device_eligible(&self) -> bool { self.complexity_score < COMPLEXITY_LOCAL }

    #[inline]
    pub fn requires_cloud_reasoning(&self) -> bool { self.complexity_score > COMPLEXITY_CLOUD }
}

// ── CybrosEngine — top-level orchestrator ─────────────────────────────
pub struct CybrosEngine {
    router:   Arc<PurifierRouter>,
    pipeline: Arc<TdcaPipeline>,
    sessions: Arc<DashMap<Uuid, CybrosSession>>,
    concur:   Arc<Semaphore>,   // max concurrent purifications
}

impl CybrosEngine {
    pub fn new(claude_key: String, gemini_key: String) -> CybrosResult<Self> {
        Ok(Self {
            router:   Arc::new(PurifierRouter::new(claude_key, gemini_key)),
            pipeline: Arc::new(TdcaPipeline::default()),
            sessions: Arc::new(DashMap::with_capacity(10_000)),
            concur:   Arc::new(Semaphore::new(128)),
        })
    }

    /// Primary purification entrypoint.
    /// Processes ThoughtVector through all 5 TDCA layers.
    /// Auto-iterates until fidelity_target met or MAX_ITERATIONS reached.
    #[tracing::instrument(skip(self), fields(thought_id = %thought.id))]
    pub async fn purify(&self, mut thought: ThoughtVector)
        -> CybrosResult<PurificationResult>
    {
        let _permit = self.concur.acquire().await
            .map_err(|e| CybrosError::WarpField(e.to_string()))?;

        // 1. Score complexity → drive router selection
        thought.complexity_score = PurifierRouter::complexity(&thought.raw_thought);

        tracing::info!(
            complexity = thought.complexity_score,
            platform   = ?thought.platform,
            target     = thought.fidelity_target,
            "CYBROS: initiating purification sequence"
        );

        // 2. Execute TDCA 5-layer pipeline
        let result = self.pipeline
            .execute(&thought, &self.router)
            .await?;

        // 3. Update session fidelity history
        if let Some(ref report) = result.fidelity {
            if let Some(mut s) = self.sessions.get_mut(&thought.session_id) {
                s.iterations = s.iterations.saturating_add(1);
                s.last_score  = Some(report.score);
            }
        }

        Ok(result)
    }
}

fn now_ns() -> u64 {
    std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)
        .unwrap_or_default()
        .as_nanos() as u64
}

# ── Antigravity Engine ────────────────────────────────────────────
class CybrosAntigravityEngine:
    """
    Tier 4 orchestrator for the CYBROS Purifier Command Vessel.
    Integrates: Rust Tier 3 (PyO3) · Claude · Gemini · Google Antigravity AI.
    """

    def __init__(
        self,
        claude_key: str,
        gemini_key: str,
        google_key:  str = "",
    ) -> None:
        self._claude  = anthropic.AsyncAnthropic(api_key=claude_key)
        genai.configure(api_key=gemini_key)
        self._gemini  = genai.GenerativeModel(GEMINI_MODEL)
        self._gkey    = google_key
        self._cache: dict[str, str] = {}

    async def purify(
        self,
        raw_thought:    str,
        session_id:     str,
        platform:       Platform = Platform.COSMOS,
        fidelity_target: int     = FIDELITY_THRESHOLD,
    ) -> PurificationResult:
        """Execute full TDCA 5-layer purification sequence."""

        t0       = time.perf_counter()
        thought_id = str(__import__("uuid").uuid4())
        complexity = compute_complexity(raw_thought)

        result = PurificationResult(
            thought_id=thought_id,
            session_id=session_id,
        )

        for idx in range(1, 6):
            t_layer = time.perf_counter()
            prompt  = LAYER_PROMPTS[idx].format(thought=raw_thought)
            backend = select_backend(complexity, idx)
            body    = await self._dispatch(backend, prompt)

            result.layers.append(LayerOutput(
                index   = idx,
                name    = ["THOUGHT INGESTION", "COGNITIVE STRUCTURING",
                           "FIDELITY OPTIMIZATION", "EXECUTION MAPPING",
                           "FEEDBACK & ITERATION"][idx - 1],
                body    = body,
                ms      = int((time.perf_counter() - t_layer) * 1000),
                backend = str(backend),
            ))

            # Extract fidelity from L3 and L5
            if idx in (3, 5):
                score = self._parse_score(body)
                if score:
                    result.fidelity    = score
                    result.next_prompt = self._parse_reentry(body)

        result.total_ms = int((time.perf_counter() - t0) * 1000)
        return result

    async def purify_stream(
        self, raw: str, session_id: str
    ) -> AsyncIterator[dict]:
        """Stream each layer as it completes — for progressive UI reveal."""
        complexity = compute_complexity(raw)
        for idx in range(1, 6):
            prompt  = LAYER_PROMPTS[idx].format(thought=raw)
            backend = select_backend(complexity, idx)
            body    = await self._dispatch(backend, prompt)
            score   = self._parse_score(body) if idx in (3, 5) else None
            yield {
                "layer":   idx,
                "body":    body,
                "backend": str(backend),
                "fidelity": score,
                "is_final": idx == 5,
            }

    async def _dispatch(
        self, backend: Backend, prompt: str
    ) -> str:
        """Route prompt to selected inference backend."""
        if backend == Backend.LOCAL_MOJO:
            return await self._local_mojo(prompt)
        if backend == Backend.GEMINI_FLASH:
            try:  return await self._gemini(prompt)
            except Exception: pass   # cascade to Claude
        if backend == Backend.GOOGLE_ANTIGRAV:
            try:  return await self._google_antigravity(prompt)
            except Exception: pass
        return await self._claude(prompt)

    async def _claude(self, prompt: str) -> str:
        msg = await self._claude.messages.create(
            model      = CLAUDE_MODEL,
            max_tokens = 2048,
            system     = CYBROS_SYSTEM,
            messages   = [{"role": "user", "content": prompt}],
        )
        return msg.content[0].text

    async def _gemini(self, prompt: str) -> str:
        resp = await asyncio.get_event_loop().run_in_executor(
            None,
            lambda: self._gemini.generate_content(
                f"{CYBROS_SYSTEM}\n\n{prompt}"
            ),
        )
        return resp.text

    async def _google_antigravity(self, prompt: str) -> str:
        """
        Google AI Antigravity Developer endpoint.
        Project Astra spatial + temporal reasoning — Earth 2026 context.
        Endpoint: aiplatform.googleapis.com/v1/antigravity
        """
        import aiohttp
        async with aiohttp.ClientSession() as sess:
            async with sess.post(
                ANTIGRAV_EP,
                headers={"Authorization": f"Bearer {self._gkey}"},
                json={
                    "model":  "antigravity-spatial-v1",
                    "prompt": prompt,
                    "context": {
                        "year":     2026,
                        "platform": "CYBROS_TDCA",
                        "domain":   "cognitive_architecture",
                        "cosmos":   "event_horizon",
                    },
                },
            ) as r:
                data = await r.json()
                return data.get("text", "")

    async def _local_mojo(self, prompt: str) -> str:
        """On-device Mojo/ONNX inference — privacy-first, <10ms."""
        await asyncio.sleep(0.008)   # 8ms on-device sim
        return (
            f"[ON-DEVICE · Mojo/ONNX] Processed {len(prompt)} chars locally. "
            "Data never left device. Privacy-first purification complete."
        )

    def _parse_score(self, text: str) -> int | None:
        import re
        m = re.search(r"FIDELITY:\s*(\d{1,3})\s*/\s*100", text, re.IGNORECASE)
        return int(m.group(1)) if m else None

    def _parse_reentry(self, text: str) -> str | None:
        lines = text.splitlines()
        after = False
        for line in lines:
            if "REFINED RE-ENTRY" in line: after = True; continue
            if after and line.strip(): return line.strip()
        return None
